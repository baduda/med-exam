# med-exam Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a pipeline that turns Polish maxillofacial-surgery PDFs into a ~1000-question MCQ bank, plus a static web app to practice it, for the Polish LDEW/nostryfikacja exam.

**Architecture:** Python does deterministic work (extract → chunk → assemble/validate). The Claude Code agent generates MCQ in-session from chunk files (no Anthropic API key). A vanilla-JS static app serves `questions.json` on GitHub Pages.

**Tech Stack:** Python 3.11, pymupdf (fitz), stdlib `json`/`re`/`unittest`; vanilla HTML/CSS/JS; GitHub Pages.

## Global Constraints

- Exam content (question/options/explanation/topic) is **Polish**; all code, comments, docs, JSON keys, identifiers are **English**.
- MCQ format: exactly 5 options A–E, one correct (`correct` ∈ {A,B,C,D,E}).
- Question id format: `t{2|3}-ch{NN}-{NNN}` (book, zero-padded chapter, running number).
- No Anthropic API: generation is done by the agent in-session, one chapter/run (~50 Q, flex with size), grounded in chunk text, resumable via `data/state.json`.
- Every question carries a real `source.pages` range from its chunk. No invented references.
- No build step for the web app. No backend. Out of scope: exam/timed mode, LLM review pass.
- `books/` and `data/chunks`, `data/questions` are gitignored (already set).

## File structure

- `pipeline/schema.py` — question schema constants + `validate_question(q)` / `validate_bank(list)`.
- `pipeline/extract.py` — PDF → `data/text/<book>.json` (chapters with page ranges, dehyphenated).
- `pipeline/chunk.py` — chapter text → `data/chunks/<book>-ch<NN>-<NNN>.json` + init `data/state.json`.
- `pipeline/assemble.py` — merge `data/questions/*.json` → validate → `data/questions.json` + copy to `web/`.
- `pipeline/tests/test_*.py` — unit tests (stdlib unittest).
- `web/index.html`, `web/app.js`, `web/style.css` — the quiz app.

```python
# pipeline/schema.py
"""Question schema and validation for the med-exam MCQ bank."""

OPTION_KEYS = ("A", "B", "C", "D", "E")


def _nonempty(v) -> bool:
    return isinstance(v, str) and v.strip() != ""


def validate_question(q: dict) -> list[str]:
    """Return a list of human-readable error strings; empty means valid."""
    errors: list[str] = []
    if not _nonempty(q.get("id")):
        errors.append("id: empty or missing")
    for field in ("topic", "question", "explanation"):
        if not _nonempty(q.get(field)):
            errors.append(f"{field}: empty or missing")
    opts = q.get("options")
    if not isinstance(opts, dict) or tuple(sorted(opts)) != OPTION_KEYS:
        errors.append("options: must be exactly keys A-E")
    else:
        for k in OPTION_KEYS:
            if not _nonempty(opts[k]):
                errors.append(f"option {k}: empty")
    if q.get("correct") not in OPTION_KEYS:
        errors.append("correct: must be one of A-E")
    src = q.get("source")
    if not isinstance(src, dict) or not _nonempty(src.get("book")):
        errors.append("source.book: empty or missing")
    if not isinstance(src, dict) or not isinstance(src.get("pages"), list) or not src.get("pages"):
        errors.append("source.pages: must be a non-empty list")
    return errors


def validate_bank(questions: list[dict]) -> list[str]:
    """Validate every question and flag duplicate ids."""
    errors: list[str] = []
    seen = set()
    for q in questions:
        qid = q.get("id", "<no-id>")
        for e in validate_question(q):
            errors.append(f"{qid}: {e}")
        if qid in seen:
            errors.append(f"{qid}: duplicate id")
        seen.add(qid)
    return errors
```

```python
# pipeline/extract.py
"""Extract cleaned, chapter-split text from the source PDFs."""
import json
import re
import sys
from pathlib import Path

import fitz  # pymupdf

BOOKS_DIR = Path("books")
OUT_DIR = Path("data/text")
# Chapter heading at line start: "9. Title" or "12.3. Title" — we treat the
# top-level "N. Title" as chapter boundary.
HEADING_RE = re.compile(r"^(\d+)\.\s+(.+)$", re.MULTILINE)


def dehyphenate(text: str) -> str:
    text = re.sub(r"-\n", "", text)          # join hyphenated line breaks
    text = re.sub(r"\n+", " ", text)          # remaining newlines -> spaces
    return re.sub(r"[ \t]+", " ", text).strip()


def split_chapters(pages: list[dict]) -> list[dict]:
    """Group pages into chapters by the first top-level 'N. Title' heading seen."""
    chapters: list[dict] = []
    current = None
    for pg in pages:
        m = HEADING_RE.search(pg["text"])
        if m:
            if current:
                chapters.append(current)
            current = {"chapter": m.group(1), "title": m.group(2).strip(),
                       "pages": [pg["page"], pg["page"]], "text": ""}
        if current is None:
            current = {"chapter": "0", "title": "front", "pages": [pg["page"], pg["page"]], "text": ""}
        current["pages"][1] = pg["page"]
        current["text"] += "\n" + pg["text"]
    if current:
        chapters.append(current)
    for c in chapters:
        c["text"] = dehyphenate(c["text"])
    return chapters


def extract_book(pdf_path: Path) -> dict:
    doc = fitz.open(pdf_path)
    pages = [{"page": i + 1, "text": doc[i].get_text()} for i in range(doc.page_count)]
    book = pdf_path.stem.split(".")[0].strip()  # "Tom2. Mansur Rahnama" -> "Tom2"
    return {"book": book, "chapters": split_chapters(pages)}


def main() -> None:
    OUT_DIR.mkdir(parents=True, exist_ok=True)
    for pdf in sorted(BOOKS_DIR.glob("*.pdf")):
        data = extract_book(pdf)
        out = OUT_DIR / f"{data['book']}.json"
        out.write_text(json.dumps(data, ensure_ascii=False, indent=1), encoding="utf-8")
        print(f"{pdf.name}: {len(data['chapters'])} chapters -> {out}")


if __name__ == "__main__":
    sys.exit(main())
```

```python
# pipeline/chunk.py
"""Split chapter text into ~700-word chunks and write chunk files + state."""
import json
import re
import sys
from pathlib import Path

TEXT_DIR = Path("data/text")
CHUNK_DIR = Path("data/chunks")
STATE_FILE = Path("data/state.json")


def chunk_text(text: str, target_words: int = 700) -> list[str]:
    paras = [p.strip() for p in re.split(r"\n\s*\n|(?<=[.?!])\s{2,}", text) if p.strip()]
    if not paras:
        paras = [text.strip()]
    chunks, buf, count = [], [], 0
    for p in paras:
        w = len(p.split())
        if count + w > target_words and buf:
            chunks.append(" ".join(buf))
            buf, count = [], 0
        buf.append(p); count += w
    if buf:
        chunks.append(" ".join(buf))
    return chunks or [text.strip()]


def main() -> None:
    CHUNK_DIR.mkdir(parents=True, exist_ok=True)
    state = {"chapters": {}}
    for text_file in sorted(TEXT_DIR.glob("*.json")):
        data = json.loads(text_file.read_text(encoding="utf-8"))
        book = data["book"]
        for ch in data["chapters"]:
            cid = f"{book.lower()}-ch{int(ch['chapter']):02d}"
            parts = chunk_text(ch["text"])
            for i, part in enumerate(parts, 1):
                chunk = {"chunk_id": f"{cid}-{i:03d}", "book": book,
                         "chapter": ch["chapter"], "topic": ch["title"],
                         "pages": ch["pages"], "text": part}
                (CHUNK_DIR / f"{chunk['chunk_id']}.json").write_text(
                    json.dumps(chunk, ensure_ascii=False, indent=1), encoding="utf-8")
            state["chapters"][cid] = {"title": ch["title"], "chunks": len(parts),
                                      "generated": False}
    STATE_FILE.write_text(json.dumps(state, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"wrote chunks for {len(state['chapters'])} chapters -> {CHUNK_DIR}")


if __name__ == "__main__":
    sys.exit(main())
```

```markdown
# Generation protocol (one chapter per run)

1. Pick the next chapter in `data/state.json` with `generated: false`. Announce which.
2. Read all its chunk files `data/chunks/<cid>-*.json`.
3. For each chunk, write single-best-answer MCQ grounded ONLY in that chunk's text:
   - 5 options A–E, exactly one correct, 4 plausible Polish distractors.
   - Everything Polish: question, options, explanation, topic (= chunk topic).
   - `source.pages` = the chunk's `pages`. Never invent page numbers.
   - id = `t{2|3}-ch{NN}-{NNN}` continuing the chapter's running count.

4. Target ~50 questions for the chapter (fewer for a short chapter — do NOT pad
   or duplicate). Skip figure-only chunks with too little text to ground a question.
5. Write the list to `data/questions/<cid>.json`.
6. Run `python pipeline/assemble.py` — fix any validation errors it reports.
7. Set `generated: true` for the chapter in `data/state.json`. Commit.
8. Stop. Next run repeats from step 1 (resumable across sessions).
```

```python
# pipeline/assemble.py
"""Merge per-chapter question files into the validated final bank."""
import json
import sys
from pathlib import Path

from pipeline.schema import validate_bank

Q_DIR = Path("data/questions")
OUT = Path("data/questions.json")
WEB_COPY = Path("web/questions.json")


def load_bank(directory: Path) -> list[dict]:
    bank: list[dict] = []
    for f in sorted(directory.glob("*.json")):
        bank.extend(json.loads(f.read_text(encoding="utf-8")))
    return bank


def main() -> int:
    bank = load_bank(Q_DIR)
    errors = validate_bank(bank)
    if errors:
        print(f"{len(errors)} validation errors:")
        for e in errors[:50]:
            print("  -", e)
        return 1
    payload = json.dumps(bank, ensure_ascii=False, indent=1)
    OUT.write_text(payload, encoding="utf-8")
    WEB_COPY.parent.mkdir(parents=True, exist_ok=True)
    WEB_COPY.write_text(payload, encoding="utf-8")
    print(f"OK: {len(bank)} questions -> {OUT} and {WEB_COPY}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```javascript
const el = (id) => document.getElementById(id);

const load = () => JSON.parse(localStorage.getItem(KEY) || '{"done":[],"correct":0,"answered":0}');

function shuffle(a) { // Fisher–Yates
  for (let i = a.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [a[i], a[j]] = [a[j], a[i]]; }

  }
}

function choose(q, k, li) {
  [...el("options").children].forEach(c => { c.onclick = null; c.classList.add("locked"); });
  const right = q.correct;
  el("options").children[["A","B","C","D","E"].indexOf(right)].classList.add("correct");
  if (k !== right) li.classList.add("wrong");
  answered++; if (k === right) correct++;
  el("explanation").textContent = q.explanation; el("explanation").hidden = false;
  el("source").textContent = `Źródło: ${q.source.book}, s. ${q.source.pages.join("–")}`; el("source").hidden = false;
  el("next-btn").hidden = false;
  const s = load(); s.answered = answered; s.correct = correct; if (!s.done.includes(q.id)) s.done.push(q.id); save(s);
  renderScore();
}

function fillTopics() {
  const topics = [...new Set(questions.map(q => q.topic))].sort();
  el("topic").innerHTML = topics.map(t => `<option>${t}</option>`).join("");
}

el("mode").onchange = () => { el("topic-wrap").hidden = el("mode").value !== "topic"; };
el("start-btn").onclick = () => { queue = buildQueue(); idx = 0; el("start").hidden = true; el("quiz").hidden = false; renderQuestion(); };
el("next-btn").onclick = () => { idx++; renderQuestion(); };
el("reset-btn").onclick = () => { save({done:[],correct:0,answered:0}); answered = 0; correct = 0; renderScore(); };

fetch("questions.json").then(r => r.json()).then(data => {
  questions = data; fillTopics();
  const s = load(); answered = s.answered; correct = s.correct; renderScore();
});
```
